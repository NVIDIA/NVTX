76/* Config string format (used by `nvtxwSessionAttributes_t::configString`):
77 *
78 * The format is key=value pairs, delimited by new-line characters or | (pipe)
79 * characters. Values are prohibited from containing those characters. If an
80 * entry begins with #, the entry (up to the next new-line or pipe) is discarded
81 * as a comment.
82 *
83 * If a config specifies the same key multiple times, only the first appearance
84 * should be honored, and the subsequent appearances should be ignored. This
85 * allows a simple scan for a particular key to loop from the beginning until
86 * the first occurrence is found, and not have to loop through the rest for
87 * repeats. Note that this means building a map from keys to values should not
88 * overwrite existing values if a found key already exists in the map. This
89 * guarantee allows adding extra key/value pairs to a config string by
90 * prepending (to override existing keys) or appending (to set values only if
91 * they weren't set already). Keys are tool-specific.
92 */
93

128NVTX_INLINE_STATIC
129void nvtxwConsumeConfigString(const char* config, nvtxwKeyValuePairConsumer_t consumer, void* state)
130{
131 const char* curRead = config;
132 const char* const lineBreak = "|\n\r";
133 const char* const whitespace = " \t\v"; /* Not including lineBreak characters */
134 int consumerStopRequested = 0;
135
136 if (!config || !consumer) return;
137
138 while (*curRead && !consumerStopRequested)
139 {
140 const char* lineBegin;
141 const char* lineEnd;
142 const char* keyBegin;
143 const char* keyEnd;
144 const char* valBegin;
145 const char* valEnd;
146
147 /* Read a line, trimming leading whitespace - get pointers to begin/end */
148 lineBegin = curRead + strspn(curRead, whitespace);
149 lineEnd = lineBegin + strcspn(lineBegin, lineBreak);
150
151 /* Set read pointer to beginning of next line, so we can continue any time */
152 curRead = lineEnd + strspn(lineEnd, lineBreak);
153
154 /* Ignore line if it's only whitespace */
155 if (lineBegin == lineEnd) continue;
156 /* Ignore line if it's a comment */
157 if (*lineBegin == '#') continue;
158
159 /* Determine if line has a key and value delimited by '=' */
160 keyBegin = lineBegin;
161 keyEnd = keyBegin;
162 while (keyEnd < lineEnd && *keyEnd != '=') ++keyEnd;
163
164 /* Ignore line if there's no '=' in the line */
165 if (keyEnd == lineEnd) continue;
166 /* Ignore line if there's no key name before '=' */
167 if (keyEnd == keyBegin) continue;
168
169 /* keyEnd now points at '=' after the key */
170 valBegin = keyEnd + 1;
171 valBegin += strspn(valBegin, whitespace);
172
173 /* Ignore line if all characters after '=' are whitespace */
174 if (valBegin == lineEnd) continue;
175
176 valEnd = lineEnd;
177
178 /* Got begin/end pointers for key and value. We know there are non-
179 * whitespace characters in both of them, and their leading whitespace
180 * was already trimmed. Now trim their trailing whitespace. */
181 while (strchr(whitespace, *(keyEnd - 1))) --keyEnd;
182 while (strchr(whitespace, *(valEnd - 1))) --valEnd;
183
184 /* Now key and value begin/end pointers can be passed to the consumer */
185 consumerStopRequested = consumer(state, keyBegin, keyEnd, valBegin, valEnd);
186 }
187}
