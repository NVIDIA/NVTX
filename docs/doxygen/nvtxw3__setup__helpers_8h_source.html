103NVTX_INLINE_STATIC
104nvtxwResultCode_t nvtxwLoadInterface(
105 const char* library,
106 const nvtxwInterface_t** interfaceOut,
107 nvtxwModuleHandle_t* moduleOut)
108{
109 nvtxwGetInterface_t getInterface;
110 const void* iface;
111 nvtxwModuleHandle_t module;
112 nvtxwResultCode_t result;
113
114 if (!interfaceOut) return NVTXW_RESULT_INVALID_ARGUMENT;
115
116 *interfaceOut = NVTX_NULLPTR;
117 if (moduleOut) *moduleOut = NVTX_NULLPTR;
118
119 getInterface = NVTX_NULLPTR;
120 module = NVTX_NULLPTR;
121 result = nvtxwLoad(
122 library,
123 &getInterface,
124 &module);
125 if (result != NVTXW_RESULT_SUCCESS) return result;
126 if (!getInterface)
127 {
128 nvtxwUnload(module);
129 return NVTXW_RESULT_LIBRARY_LOAD_FAILED;
130 }
131
132 iface = NVTX_NULLPTR;
133 result = getInterface(NVTXW_INTERFACE_VERSION, &iface);
134 if (result != NVTXW_RESULT_SUCCESS || !iface)
135 {
136 nvtxwUnload(module);
137 return result != NVTXW_RESULT_SUCCESS
138 ? result
139 : NVTXW_RESULT_INTERFACE_VERSION_NOT_SUPPORTED;
140 }
141
142 *interfaceOut = NVTX_STATIC_CAST(const nvtxwInterface_t*, iface);
143 if (moduleOut) *moduleOut = module;
144 return NVTXW_RESULT_SUCCESS;
145}

272NVTX_INLINE_STATIC
273nvtxwResultCode_t nvtxwStreamOpen(
274 const nvtxwInterface_t* iface,
275 nvtxwSessionHandle_t session,
276 const char* streamName,
277 nvtxDomainHandle_t domain,
278 const char* scopePath,
279 uint64_t timeDomainId,
280 nvtxwStreamHandle_t* streamOut)
281{
282 nvtxwStreamAttributes_t streamAttr;
283 uint64_t scopeId;
284 nvtxwResultCode_t result;
285
286 if (!iface || !iface->StreamOpen || !session || !streamOut)
287 return NVTXW_RESULT_INVALID_ARGUMENT;
288
289 *streamOut = NVTX_NULLPTR;
290
291 scopeId = NVTX_SCOPE_NONE;
292 if (scopePath && *scopePath)
293 {
294 nvtxDomainHandle_t scopeDomain = domain;
295 if (!scopeDomain)
296 {
297 result = nvtxwDomainRegister(
298 iface,
299 session,
300 NVTX_NULLPTR,
301 &scopeDomain);
302 if (result != NVTXW_RESULT_SUCCESS) return result;
303 }
304
305 result = nvtxwScopeRegister(
306 iface,
307 scopeDomain,
308 scopePath,
309 &scopeId);
310 if (result != NVTXW_RESULT_SUCCESS) return result;
311 }
312
313 result = nvtxwStreamAttributesInit(&streamAttr);
314 if (result != NVTXW_RESULT_SUCCESS) return result;
315
316 streamAttr.name = streamName;
317 streamAttr.domain = domain;
318 streamAttr.scopeId = scopeId;
319 streamAttr.timeDomainId = timeDomainId;
320
321 return iface->StreamOpen(session, &streamAttr, streamOut);
322}

338nvtxwResultCode_t nvtxwEventWriterOpen(
339 const nvtxwInterface_t* iface,
340 nvtxwSessionHandle_t session,
341 const char* streamName,
342 nvtxDomainHandle_t domain,

356
357 result = nvtxwEventSchemasRegister(

359 domain,

363
364 stream = NVTX_NULLPTR;
365 result = nvtxwStreamOpen(

367 session,
368 streamName,
369 domain,
370 scopePath,
371 timeDomainId,
372 &stream);
373 if (result != NVTXW_RESULT_SUCCESS) return result;
374
375 return nvtxwEventWriterInit(writerOut, iface, stream, &schemaIds);
376}
