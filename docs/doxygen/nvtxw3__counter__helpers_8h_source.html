42NVTX_INLINE_STATIC
43nvtxwResultCode_t nvtxwCounterRegister(
44 const nvtxwInterface_t* iface,
45 nvtxDomainHandle_t domain,
46 const char* name,
47 uint64_t schemaId,
48 const nvtxSemanticsHeader_t* semantics,
49 uint64_t* counterIdOut)
50{
51 nvtxCounterAttr_t counterAttr;
52
53 if (!iface || !iface->CounterRegister || !domain ||
54 !schemaId || !counterIdOut)
55 return NVTXW_RESULT_INVALID_ARGUMENT;
56
57 *counterIdOut = NVTX_COUNTER_ID_NONE;
58
59 memset(&counterAttr, 0, sizeof(counterAttr));
60 counterAttr.structSize = sizeof(counterAttr);
61 counterAttr.schemaId = schemaId;
62 counterAttr.name = name;
63 counterAttr.semantics = semantics;
64
65 return iface->CounterRegister(domain, &counterAttr, counterIdOut);
66}
