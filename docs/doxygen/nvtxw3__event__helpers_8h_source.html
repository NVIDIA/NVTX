578 /* The standalone UTF-8 message schema is shared by all UTF-8 event schemas,
579 * so it is registered whenever any UTF-8 schema is selected. */
580 const uint32_t utf8Selectors =
581 NVTXW_EVENT_HELPER_SCHEMA_MARK_UTF8 |
582 NVTXW_EVENT_HELPER_SCHEMA_RANGE_PUSHPOP_UTF8 |
583 NVTXW_EVENT_HELPER_SCHEMA_RANGE_STARTEND_UTF8 |
584 NVTXW_EVENT_HELPER_SCHEMA_RANGE_PUSH_UTF8 |
585 NVTXW_EVENT_HELPER_SCHEMA_RANGE_START_UTF8;

587 if (!iface || !iface->SchemaRegister || !domain || !schemaIdsOut || !schemaMask)
588 return NVTXW_RESULT_INVALID_ARGUMENT;

590 return NVTXW_RESULT_INVALID_ARGUMENT;

594#define NVTXW_EVENT_HELPER_SCHEMA_DESC( \
595 INDEX, SELECTOR, FLAGS, TYPE, ENTRIES, PAYLOAD_TYPE, SCHEMA_ID_FIELD) \
596 do \
597 { \
598 schemas[INDEX].selector = (SELECTOR); \
599 schemas[INDEX].flags = (FLAGS); \
600 schemas[INDEX].type = (TYPE); \
601 schemas[INDEX].entries = (ENTRIES); \
602 schemas[INDEX].numEntries = sizeof(ENTRIES) / sizeof((ENTRIES)[0]); \
603 schemas[INDEX].payloadStaticSize = sizeof(PAYLOAD_TYPE); \
604 schemas[INDEX].schemaId = &schemaIdsOut->SCHEMA_ID_FIELD; \
605 } while (0)

651 schemas[12].type = NVTX_PAYLOAD_SCHEMA_TYPE_DYNAMIC;
652 schemas[12].entries = eventMessageUtf8Entries;
653 schemas[12].numEntries =
654 sizeof(eventMessageUtf8Entries) / sizeof(eventMessageUtf8Entries[0]);
655 schemas[12].payloadStaticSize = 0;
656 schemas[12].schemaId = &schemaIdsOut->eventMessageUtf8;
657
658 memset(&attr, 0, sizeof(attr));
659 attr.fieldMask =
660 NVTX_PAYLOAD_SCHEMA_ATTR_FIELD_TYPE |
661 NVTX_PAYLOAD_SCHEMA_ATTR_FIELD_FLAGS |
662 NVTX_PAYLOAD_SCHEMA_ATTR_FIELD_ENTRIES |
663 NVTX_PAYLOAD_SCHEMA_ATTR_FIELD_NUM_ENTRIES |
664 NVTX_PAYLOAD_SCHEMA_ATTR_FIELD_STATIC_SIZE;
665
666 for (i = 0; i < sizeof(schemas) / sizeof(schemas[0]); ++i)
667 {
668 if (!(schemaMask & schemas[i].selector)) continue;
669
670 /* The UTF-8 message schema embeds a variable-length string and is
671 * therefore dynamic; all other schemas are static. */
672 attr.type = schemas[i].type;
673 attr.flags = schemas[i].flags;
674 attr.entries = schemas[i].entries;
675 attr.numEntries = schemas[i].numEntries;
676 attr.payloadStaticSize = schemas[i].payloadStaticSize;
677 result = iface->SchemaRegister(domain, &attr, schemas[i].schemaId);
678 if (result != NVTXW_RESULT_SUCCESS) return result;
679 }
680
681 return NVTXW_RESULT_SUCCESS;
682}

705/* Write a UTF-8 event as two payloads: the fixed event header and a message
706 * payload whose bytes point directly at the caller's message string. This avoids
707 * deep copy, heap allocation, and copying the message into a combined buffer.
708 *
709 * `messageLength` is the message length in bytes, or 0 to derive it with
710 * `strlen`. A non-zero length supports non-null-terminated strings and is
711 * never read past, so the message need not be null-terminated. A NULL message
712 * is sent as an empty (present, zero-length) message so the event always
713 * carries a message. */
714NVTX_INLINE_STATIC
715nvtxwResultCode_t nvtxwEventHelperImplWriteUtf8(
716 const nvtxwInterface_t* iface,
717 nvtxwStreamHandle_t stream,
718 uint64_t schemaId,
719 uint64_t messageSchemaId,
720 const void* header,
721 size_t headerSize,
722 const char* message,
723 uint32_t messageLength)
724{
725 nvtxPayloadData_t payloads[2];
726 size_t length;
727
728 if (!iface || !iface->EventWrite || !stream ||
729 !schemaId || !messageSchemaId || !header || !headerSize)
730 return NVTXW_RESULT_INVALID_ARGUMENT;
731
732 if (!message)
733 length = 0;
734 else if (messageLength == 0)
735 length = strlen(message);
736 else
737 length = messageLength;
738
739 payloads[0].schemaId = schemaId;
740 payloads[0].size = headerSize;
741 payloads[0].payload = header;
742
743 /* The message bytes are the payload. Payloads of size 0 are ignored, so an
744 * empty message is sent as a single null code unit (a present empty value). */
745 payloads[1].schemaId = messageSchemaId;
746 if (length == 0)
747 {
748 payloads[1].size = 1;
749 payloads[1].payload = "";
750 }
751 else
752 {
753 payloads[1].size = length;
754 payloads[1].payload = message;
755 }
756
757 return iface->EventWrite(stream, payloads, 2);
758}
